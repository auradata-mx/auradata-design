import { Meta, ColorPalette, ColorItem } from '@storybook/addon-docs/blocks'
import { TypeScale } from './TypeScale'
import theme from '@auradata/tokens/shadcn-theme.json'

<Meta title="Brand/Tokens" />

export const pairs = (mode) =>
  Object.entries(theme[mode]).filter(
    ([name]) =>
      !/^(chart|sequential|diverging)-|-\d{2,3}$/.test(name) && !['radius', 'gradient-brand'].includes(name),
  )

# Tokens de AuraData

Cada valor de esta página sale de `@auradata/tokens`. Cambia `packages/tokens/src/tokens` y vuelve a compilar; nunca escribas un color en un componente.

## Claro

<ColorPalette>
  {pairs('light').map(([name, value]) => (
    <ColorItem key={name} title={`--${name}`} colors={{ [name]: value }} />
  ))}
</ColorPalette>
