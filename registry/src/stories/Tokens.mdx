import { Meta, ColorPalette, ColorItem } from '@storybook/addon-docs/blocks'
import { TypeScale } from './TypeScale'
import theme from '@auradata/tokens/shadcn-theme.json'

<Meta title="Brand/Tokens" />

export const pairs = (mode) =>
  Object.entries(theme[mode]).filter(([name]) => !name.startsWith('chart-') && !['radius', 'gradient-brand'].includes(name))

# AuraData tokens

Every value on this page is read from `@auradata/tokens`. Change `packages/tokens/src/tokens` and rebuild; never edit a color in a component.

## Light

<ColorPalette>
  {pairs('light').map(([name, value]) => (
    <ColorItem key={name} title={name} subtitle={`--${name}`} colors={{ [name]: value }} />
  ))}
</ColorPalette>
